if (!function_exists('format_speed')) {
    function format_speed($hours) {
        if ($hours < 24) return $hours . ' ساعة';
        $days = floor($hours / 24);
        $rem_hours = $hours % 24;
        return $days . ' يوم' . ($rem_hours > 0 ? ' و ' . $rem_hours . ' ساعة' : '');
    }
}
